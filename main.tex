\cvsection{Research Experience}
\cvrow{\textbf{Intelligent Control and Estimation of Things Lab}, NEU}{July 2024 - Present}
\cvresearchdetails{Advisor: Prof. Sze Zheng Yong}{Graduate Research Assistant}
\cvproject{\textbf{\textit{CAffNet: Hard Constraint-Affine Neural Networks}}}
\begin{itemize}
  \item Proposed CAffNet, a closed-form neural network framework that guarantees hard satisfaction of an arbitrary number of input-dependent affine inequality constraints without requiring full row rank assumptions.
  \item Proposed a constraint decomposition with a trainable projection module to handle linearly dependent constraints and non-unique projections.
  \item Evaluated CAffNet with various experiments instantiated with feedforward and transformer architectures, achieving zero constraint violations and better performance.
\end{itemize}
\cvproject{\textbf{\textit{Learning Safe-by-Design Neural Network Controllers}}}
\begin{itemize}
  \item Proposed a safe-by-design neural network control framework that integrates trainable CBFs with CAffNet, enabling joint learning of the controller and CBF parameters with constraint satisfaction guaranteed by construction and preserving the expressiveness of control outputs while eliminating the need for a separate safety filtering optimization.
  \item Introduced a computationally efficient formulation of CAffNet that significantly reduces the time cost of the projection step.
  \item Demonstrated the effectiveness of our approach through extensive simulations, showing improved safety enforcement and control performance.
\end{itemize}
